export function ScoreRing({ score = 0, label, size = 120, color = 'var(--state-optimal)', strokeWidth = 12 }) {
  const r = (size - strokeWidth) / 2 - 3;
  const c = 2 * Math.PI * r;
  const clamped = Math.max(0, Math.min(100, score));
  const offset = c * (1 - clamped / 100);
  const numSize = Math.round(size * 0.28);
  return (
    <div style={{ position: 'relative', width: size, height: size, flexShrink: 0, fontFamily: 'var(--font-sans)' }}>
      <svg width={size} height={size} viewBox={`0 0 ${size} ${size}`} style={{ transform: 'rotate(-90deg)' }}>
        <circle cx={size / 2} cy={size / 2} r={r} fill="none" stroke="var(--mint)" strokeWidth={strokeWidth} />
        <circle
          cx={size / 2}
          cy={size / 2}
          r={r}
          fill="none"
          stroke={color}
          strokeWidth={strokeWidth}
          strokeLinecap="round"
          strokeDasharray={c}
          strokeDashoffset={offset}
          style={{ transition: 'stroke-dashoffset .6s ease' }}
        />
      </svg>
      <div style={{ position: 'absolute', inset: 0, display: 'flex', flexDirection: 'column', alignItems: 'center', justifyContent: 'center' }}>
        <div style={{ fontSize: numSize, fontWeight: 'var(--fw-bold)', letterSpacing: 'var(--ls-tight)', lineHeight: 1, color: 'var(--text-body)' }}>
          {Math.round(clamped)}
        </div>
        {label && (
          <div style={{ fontSize: 'var(--fs-label)', fontWeight: 'var(--fw-semibold)', color: 'var(--text-muted)', marginTop: 3 }}>{label}</div>
        )}
      </div>
    </div>
  );
}
