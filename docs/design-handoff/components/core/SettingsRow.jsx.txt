import { Toggle } from './Toggle.jsx';

export function SettingsRow({ label, description, value, control = 'chevron', on = false, first = false, onToggle, onClick }) {
  return (
    <div
      onClick={control === 'chevron' ? onClick : undefined}
      style={{
        display: 'flex',
        alignItems: 'center',
        gap: 12,
        padding: 'var(--sp-row) 2px',
        borderTop: first ? 'none' : '1px solid var(--border-hairline)',
        fontFamily: 'var(--font-sans)',
        cursor: control === 'chevron' ? 'pointer' : 'default',
      }}
    >
      <div style={{ flex: 1, minWidth: 0 }}>
        <div style={{ fontSize: 'var(--fs-control)', fontWeight: 'var(--fw-semibold)', color: 'var(--text-body)' }}>{label}</div>
        {description && (
          <div style={{ fontSize: 'var(--fs-caption)', color: 'var(--text-muted)', marginTop: 2, lineHeight: 1.4 }}>{description}</div>
        )}
      </div>
      {value && (
        <span style={{ fontSize: 'var(--fs-body)', color: 'var(--text-faint)', whiteSpace: 'nowrap' }}>{value}</span>
      )}
      {control === 'toggle' && <Toggle on={on} onChange={onToggle} ariaLabel={label} />}
      {control === 'chevron' && (
        <svg viewBox="0 0 24 24" style={{ width: 16, height: 16, stroke: 'var(--text-faint)', strokeWidth: 2, fill: 'none', strokeLinecap: 'round', strokeLinejoin: 'round', flexShrink: 0 }}>
          <path d="M9 6l6 6-6 6" />
        </svg>
      )}
    </div>
  );
}
